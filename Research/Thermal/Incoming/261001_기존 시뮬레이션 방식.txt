def load_meas(g):
    blocks, cur = {}, None
    for ln in open(MEAS):
        s = ln.split()
        if not s: continue
        if s[0].lower().startswith("gap"):
            cur = int(''.join(c for c in ''.join(s) if c.isdigit())); blocks[cur] = []
        elif cur is not None and len(s) == 9:
            blocks[cur].append([float(x) for x in s])
    a = np.array(blocks[g]); f = a[:, 0]
    s11db = a[:, 1]  # measured S11 magnitude already in dB (col index 1)
    return f, s11db


def load_s2p_s11db(path):
    fmt, mult, nums = "ma", 1e9, []
    unit = {"hz": 1.0, "khz": 1e3, "mhz": 1e6, "ghz": 1e9}
    for line in open(path):
        line = line.split("!")[0].strip()
        if not line: continue
        if line.startswith("#"):
            tok = line[1:].lower().split()
            for i, t in enumerate(tok):
                if t in unit: mult = unit[t]
                if t in ("ma", "db", "ri"): fmt = t
            continue
        nums += [float(x) for x in line.split()]
    nums = np.array(nums); n = len(nums) // 9; a = nums[:n * 9].reshape(n, 9); f = a[:, 0] * mult
    x, y = a[:, 1], a[:, 2]
    if fmt == "ri": s11 = x + 1j * y
    elif fmt == "db": s11 = 10 ** (x / 20) * np.exp(1j * np.deg2rad(y))
    else: s11 = x * np.exp(1j * np.deg2rad(y))
    return f / 1e9, 20 * np.log10(np.abs(s11))
